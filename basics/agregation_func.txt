* Теория: COUNT, SUM, AVG, MIN, MAX, группировка, HAVING.
* Практика: подсчёт статистики, группировка по полям, фильтрация групп.
* Задание: посчитать количество записей, среднюю зарплату по отделам, отфильтровать группы по условию.

Подготовительные этапы:

Каждый этап можно контролировать на небольшой выборке данных

table test_users limit 10;


В ранее созданную таблицу test_users добавить колонку с зп

ALTER TABLE test_users
ADD COLUMN salary NUMERIC(10, 2);

Потом заполнить сгенерированными данными

UPDATE test_users
SET salary = (random() * 120000 + 30000)::numeric(10, 2);

Добавить колонку с отделом

ALTER TABLE test_users
ADD COLUMN department VARCHAR(100);

Заполнить данными

UPDATE test_users
SET department = (
    CASE (random() * 3)::int
        WHEN 0 THEN 'IT'
        WHEN 1 THEN 'logistic'
        WHEN 2 THEN 'sales'
    END
);

-- COUNT
SELECT COUNT(*)
FROM test_users;

-- SUM
SELECT SUM(salary)
FROM test_users;

-- AVG
SELECT AVG(salary)
FROM test_users;

-- MIN
SELECT MIN(salary)
FROM test_users;

-- MAX
SELECT MAX(salary)
FROM test_users;

-- Количество записей по отделам
SELECT department,
    COUNT(*)
FROM test_users
GROUP BY department;

-- Средняя зп по отделам
SELECT department,
    AVG(salary)
FROM test_users
GROUP BY department;

-- Фильтрация групп по средней зп
SELECT department,
    AVG(salary)
FROM test_users
GROUP BY department
HAVING AVG(salary) > 90000;

-- Количество записей и средняя зп по отделам
-- со средней зп > 90000
SELECT department,
    COUNT(*),
    AVG(salary)
FROM test_users
GROUP BY department
HAVING AVG(salary) > 90000;
